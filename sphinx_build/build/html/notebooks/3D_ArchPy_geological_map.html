def topo(x,y):
    x /= 30
    y /= 30
    res = (np.sin(-(x-2)**2+1/7*(y-2)**2)) + \
    (np.sin(-(x-1)**2+1/2*(y-1)**2))+ \
    2*(np.sin(-(x-2)**2+1/2*(y-3)**2))

    #res[res<0] = 0
    return res

x  =np.linspace(x0, x1, nx)
y = np.linspace(y0, y1, ny)
X,Y = np.meshgrid(x, y)
top = topo(X, Y)
top -= 6

dimensions = (nx, ny, nz)
spacing = (sx, sy, sz)
origin = (x0, y0, z0)
